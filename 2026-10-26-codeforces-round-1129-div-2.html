Codeforces Round 1129 (Div. 2) 个人题解

链接 https://codeforces.com/contest/2277

A

取每个集合的最大值的 mex, 从小到大枚举 mex 的答案, 构造最大值为 mex 的集合即可
构造需要 mex 作为最大值, 以及 k - 1 个 <= mex 的值填充集合

void solve() {
    int n, k;
    std::cin >> n >> k;

    std::map<int, int> cnt;
    for (int _ = 0; _ < n; ++_) {
        int x;
        std::cin >> x;
        ++cnt[x];
    }

    int mex = 0, tot = 0;
    for (mex = 0; mex <= n; ++mex) {
        if (cnt[mex] == 0) break;
        --cnt[mex], tot += cnt[mex];
        if (tot < k - 1) break;
        tot -= k - 1;
    }

    std::cout << mex << '\n';
}

B

假设我们已经选出来一个 subseq, 则是否能达到目标状态, 最优的做法是指定最小值为变成 0 的选项, 然后把它变成 0, 再把其他删掉
然后显然序列越长, 最小值可以减少的值越多, 越能靠近 0, 所以直接选择原序列就好, 这样最小值最多能减少 len - 1 次
在减少 len - 1 之后还不能变成 0, 则下一次操作必然变为空序列, 达不到目标状况

void solve() {
    int n;
    std::cin >> n;

    int minv = -1;
    for (int _ = 0; _ < n; ++_) {
        int x;
        std::cin >> x;
        if (minv == -1) minv = x;
        else if (x < minv) minv = x;
    }

    if (minv <= n - 1) std::cout << n << '\n';
    else std::cout << "0\n";
}

C

在以 1 为根的树上, lca(n, u) == 1 的点 u 无论怎么与 1 连接, 都不会改变 dis(1, n)
每一次只会连接 v 和 1, 并不会改变 dis(n, v)
因此对于被连边的且 lca(n, u) != 1 的点 u, 取 dis(n, u) 的最大值 + 1 即可

void solve() {
    int n, m;
    std::cin >> n >> m;

    std::vector<std::vector<int>> t(n + 1);
    for (int _ = 0; _ < n - 1; ++_) {
        int x, y;
        std::cin >> x >> y;
        t[x].push_back(y);
        t[y].push_back(x);
    }

    std::vector<int> dis(n + 1, 0);
    std::vector<int> f(n + 1);
    auto dfs = [&](auto&& dfs, int cur, int fa, int d) -> void {
        dis[cur] = d, f[cur] = fa;
        for (auto son : t[cur]) {
            if (son == fa) continue;
            dfs(dfs, son, cur, d + 1);
        }
    };
    dfs(dfs, n, 0, 0);

    std::vector<int> lcn(n + 1, -1);
    auto dfs2 = [&](auto&& dfs2, int cur, int fa, int beg) -> void {
        if (beg == -1) {
            lcn[cur] = cur;
            for (auto son : t[cur]) {
                if (son == fa || son == f[cur]) continue;
                dfs2(dfs2, son, cur, cur);
            } 
            if (f[cur] != fa) dfs2(dfs2, f[cur], cur, -1);
        } else {
            lcn[cur] = beg;
            for (auto son : t[cur]) {
                if (son == fa) continue;
                dfs2(dfs2, son, cur, beg);
            }
        }
    };
    dfs2(dfs2, 1, 0, -1);

    int ans = dis[1];
    for (int _ = 0; _ < m; ++_) {
        int x;
        std::cin >> x;
        if (lcn[x] == 1) continue;
        ans = std::max(ans, dis[x] + 1);
    }

    std::cout << ans << '\n';
}

D1

对于每一个点的 f(n), 是有比较快速的求法的, 本人思索一番后, 决定使用以下递归求法解决

  • 其中一个答案必然是取刚好比 n 大的 mask, 然后答案是 k + (mask - n)
  • 否则求 n 减去刚好比它小的最大 mask 的函数值, 再加上形成 mask 的代价和断掉操作 1 的代价

刚好对了

void solve() {
    i64 l, r;
    std::cin >> l >> r;

    auto f = [](auto&& f, i64 x) -> i64 {
        int cnt = 0;
        for (i64 xx = x; xx; xx >>= 1, ++cnt) ;
        i64 ans = (1LL << cnt) - x + cnt - 1;
        if (cnt >= 3) {
            i64 rem = (1LL << (cnt - 1)) - 2;
            ans = std::min(ans, cnt + f(f, x - rem));
        }
        return ans;
    };

    i64 ans = 0;
    for (i64 i = l; i <= r; ++i) {
        ans = (ans + f(f, i)) % MOD;
    }
    std::cout << ans << '\n';
}

F

对于每一个点 u, 假设它对根 r 造成了贡献, 则贡献以以下方法计算

令 u 为根, r 在 u 的邻居 v 的子树上, 贡献为 all - subtree_sum(v)

那 u 什么时候对 r 会造成贡献呢? 在 u 到 r 的路径上, 没有已经被操作的节点
如果我们把操作后的节点在树上删去, 则上述条件变为 u 和 r 在一个连通块上

我们反向遍历 P, 撤回删除的节点 u (将 u 旁边没有被删除的点连起来), 则可以较为方便地使用并查集维护连通块
每次撤销 (将点加入并查集), 都需要对即将联通的集合 (每一个可能的根节点) 的答案加上 all - subtree_sum(v), 可以使用 带权并查集 较为方便地维护集合加和单点查的性质

subtree_sum(v) 可以使用任意根遍历的子树和快速求出

using i64 = long long;
constexpr i64 MOD = 998'244'353;

struct DSUW {
    int n;
    std::vector<int> fa, sz;
    std::vector<i64> tag, d;

    DSUW(int n)
    : n(n), fa(n), sz(n, 1), tag(n, 0), d(n, 0) {
        std::iota(fa.begin(), fa.end(), 0);
    }

    int find(int x) {
        if (x == fa[x]) return x;
        int f = fa[x];
        fa[x] = find(f);
        d[x] = (d[x] + d[f]) % MOD;
        return fa[x];
    }
    void merge(int x, int y) {
        int rx = find(x), ry = find(y);
        if (rx == ry) return;
        if (sz[rx] < sz[ry]) {
            d[rx] = (tag[rx] - tag[ry] + MOD) % MOD;
            fa[rx] = ry, sz[ry] += sz[rx];
        } else {
            d[ry] = (tag[ry] - tag[rx] + MOD) % MOD;
            fa[ry] = rx, sz[rx] += sz[ry];
        }
    }
    void add(int x, i64 v) {
        int rx = find(x);
        tag[rx] = (tag[rx] + v % MOD + MOD) % MOD;
    }
    i64 get(int x) {
        int rx = find(x);
        return (d[x] + tag[rx]) % MOD;
    }
};

void solve() {
    int n;
    std::cin >> n;

    std::vector<std::vector<int>> t(n + 1);
    for (int _ = 0; _ < n - 1; ++_) {
        int u, v;
        std::cin >> u >> v;
        t[u].push_back(v);
        t[v].push_back(u);
    }

    std::vector<i64> a(n + 1);
    for (int i = 1; i <= n; ++i) std::cin >> a[i];

    std::vector<int> p(n);
    for (auto& x : p) std::cin >> x;
    std::reverse(p.begin(), p.end());

    i64 all = 0;
    std::vector<int> df(n + 1, -1);
    std::vector<i64> sum(n + 1, 0);
    auto dfs = [&](auto&& dfs, int cur, int fa) -> void {
        all = (all + a[cur]) % MOD;
        sum[cur] = a[cur];
        df[cur] = fa;
        for (auto son : t[cur]) {
            if (son == fa) continue;
            dfs(dfs, son, cur);
            sum[cur] = (sum[cur] + sum[son]) % MOD;
        }
    };
    dfs(dfs, 1, 0);

    auto sbt = [&](int u, int v) {
        if (df[v] == u) return sum[v];
        return (all - sum[u] + MOD) % MOD;
    };

    DSUW dsu(n + 1);
    std::vector<bool> vis(n + 1, false);

    for (auto u : p) {
        vis[u] = true;
        dsu.add(u, all);
        for (auto v : t[u]) 
            if (vis[v]) dsu.add(v, (all - sbt(u, v) + MOD) % MOD);
        for (auto v : t[u])
            if (vis[v]) dsu.merge(v, u);
    }

    for (int i = 1; i <= n; ++i)
        std::cout << dsu.get(i) << ' ';
    std::cout << '\n';
}
